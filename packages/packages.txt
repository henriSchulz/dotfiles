# Packages this setup installs on top of a stock Omarchy install.
#
# Deliberately EXCLUDED, because the Omarchy installer already owns them:
#   - everything in /usr/share/omarchy/install/omarchy-{base,other}.packages
#   - base system / boot: efibootmgr, intel-ucode, mkinitcpio, sudo
#   - Omarchy itself: omarchy, omarchy-keyring, omarchy-settings
#
# One package per line. Comments and blank lines are ignored.

# Required by this repo. rsync is on this machine only because
# kernel-modules-hook happens to depend on it — steps 30, 40 and 80 use it
# directly, so it gets named here rather than left to chance.
stow
rsync
# xdg-open (and so `open`) detects file types with `file` unless this is
# installed; `file` calls .md text/plain, so step 25's defaults would not apply.
perl-file-mimeinfo

# Editors / dev
visual-studio-code-bin
verilator
gtkwave
blueprint-compiler
gtksourceview5

# CLI / agents
antigravity-cli
voxtype-bin
openbsd-netcat

# Themed screensaver (henri.idle -> omarchy-screensaver-themed).
# imagemagick, socat and jq are already in omarchy-base.packages.
chafa

# iCloud Photos client. gtk4 and python are already in omarchy-base.packages;
# libheif is what decodes the HEIC originals.
libadwaita
libheif

# Audio. rtkit lets PipeWire run with realtime priority (without it the
# sound crackles under load); EasyEffects with the LSP and Calf plugins
# makes up for the missing Waves MaxxAudio on the XPS 13 9310 speakers.
rtkit
easyeffects
lsp-plugins-lv2
calf

# Desktop apps. Bitwarden holds the passwords; the desktop app has no
# aarch64 build, so only the CLI is here (and the browser extension).
# KeePass only opens an older archive.
signal-desktop
bitwarden-cli
keepass

# Fonts. The SF Symbols fallback font is NOT here and never in this repo —
# it is not redistributable. See MANUAL-STEPS.md.
inter-font


# iCloud calendars in the clock plugin's popup. vdirsyncer mirrors CalDAV
# into ~/.local/share/calendars; the python trio is what
# `omarchy-calendar-events` reads that vdir with — recurrence expansion is
# the part worth a library.
vdirsyncer
python-icalendar
python-recurring-ical-events
python-dateutil

# Building my own things: hyprswipe (CMake + Ninja against the installed
# Hyprland headers), menubar (esbuild via npm). Finder needs nothing beyond
# the GTK4/python stack Omarchy already brings.
cmake
ninja
nodejs-lts-jod

# LaTeX for university work. The full texlive-* split, not texlive-most,
# so an update does not pull half a gigabyte of unused collections.
texlive-basic
texlive-latex
texlive-latexrecommended
texlive-latexextra
texlive-fontsrecommended
texlive-fontsextra
texlive-langgerman
texlive-bibtexextra
texlive-binextra
texlive-pictures
biber

# Video/audio decoding beyond the base set (kdenlive, mpv, thumbnails).
gst-plugins-good
gst-plugins-bad
gst-plugins-ugly
gst-libav
gst-plugin-va

# Hardware, power and thermals — the tools the battery and power-profile
# experiments were measured with (see ~/Projects/akku-test).
powertop
glmark2
stress-ng
smartmontools
nvme-cli

# Disks, images and rescue: the NVMe migration and the Windows backup on the
# external SSD were done with these.
ntfs-3g
ntfsprogs
7zip
wimlib
libisoburn
gptfdisk
arch-install-scripts

# Network and diagnostics
nfs-utils
wireshark-cli
fwupd
usbutils
wifite

# Misc libraries and tools pulled in by the above: libshumate (maps widget),
# qt6-connectivity (Bluetooth in Quickshell), qt6-tools, python-cryptography
# (iCloud keychain), python-fonttools (icon-font work) and rust.
libshumate
qt6-connectivity
qt6-tools
python-cryptography
python-fonttools
rust
