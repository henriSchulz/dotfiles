# Packages this setup installs on top of a stock Omarchy install.
#
# Deliberately EXCLUDED, because the Omarchy installer already owns them:
#   - everything in /usr/share/omarchy/install/omarchy-{base,other}.packages
#   - base system / boot: efibootmgr, intel-ucode, mkinitcpio, sudo
#   - Omarchy itself: omarchy, omarchy-keyring, omarchy-settings
#
# One package per line. Comments and blank lines are ignored.

# Required by this repo. rsync is on this machine only because
# kernel-modules-hook happens to depend on it — steps 30, 40 and 80 use it
# directly, so it gets named here rather than left to chance.
stow
rsync
# xdg-open (and so `open`) detects file types with `file` unless this is
# installed; `file` calls .md text/plain, so step 25's defaults would not apply.
perl-file-mimeinfo

# Editors / dev
visual-studio-code-bin
verilator
gtkwave
blueprint-compiler
gtksourceview5

# CLI / agents
antigravity-cli
voxtype-bin
openbsd-netcat

# Themed screensaver (henri.idle -> omarchy-screensaver-themed).
# imagemagick, socat and jq are already in omarchy-base.packages.
chafa

# iCloud Photos client. gtk4 and python are already in omarchy-base.packages;
# libheif is what decodes the HEIC originals.
libadwaita
libheif

# Audio. rtkit lets PipeWire run with realtime priority (without it the
# sound crackles under load); EasyEffects with the LSP and Calf plugins
# makes up for the missing Waves MaxxAudio on the XPS 13 9310 speakers.
rtkit
easyeffects
lsp-plugins-lv2
calf

# Desktop apps
signal-desktop
geary
freerdp

# Fonts
inter-font
